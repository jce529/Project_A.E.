```csharp
// Phase 11 - Save data schema (POCO).
// Decisions: D-02 (single slot), D-03 (dictionary stubs), D-03b (item list stub),
//            D-03c (player stat fields), D-05 (name-based spawn restore, not raw XY).
// Serialized by Newtonsoft.Json - Dictionary<string, bool> round-trips natively,
// which is exactly why UnityEngine.JsonUtility cannot be used here.

using System.Collections.Generic;

public class SaveData
{
    // Schema version for future migration. Bump only when field meaning changes.
    public int SaveVersion = 1;

    // D-05: location is stored as scene name + spawn point GameObject name,
    // NOT raw x/y floats. PlayerSpawner.targetSpawnPointName consumes SpawnPointName
    // and PlayerSpawner.ApplySpawn() resolves it by GameObject.name lookup.
    public string SceneName = "";
    public string SpawnPointName = "";

    // D-03c: mirrors PlayerStats/HP public getters (Health, MaxHealth, MaxTotalHealth).
    public PlayerStatsSaveData PlayerStats = new PlayerStatsSaveData();

    // D-03: stub. Key = boss id string ("TutorialBoss", "WoodBoss", "WaterSpirit",
    // "WaterMonster"). Value = defeated. Entries are added on boss death only.
    public Dictionary<string, bool> BossProgress = new Dictionary<string, bool>();

    // D-03: stub. Key = gimmick id string. Currently never written - the project has
    // no persistent map gimmicks yet. Schema exists so later phases can fill it.
    public Dictionary<string, bool> MapGimmickState = new Dictionary<string, bool>();

    // D-03b: stub. Empty list - no item/inventory system exists in the project yet.
    public List<string> Items = new List<string>();
}

public class PlayerStatsSaveData
{
    public float Health;
    public float MaxHealth;
    public float MaxTotalHealth;
}
```

주의사항:
- `MonoBehaviour` 를 상속하지 않는 순수 POCO 다. `[System.Serializable]` 어트리뷰트는 붙이지 않는다 — Newtonsoft.Json 은 필요로 하지 않으며, Unity 인스펙터에 노출할 의도가 없다.
- 필드는 전부 `public` 이어야 Newtonsoft 기본 설정에서 직렬화된다.
- 필드 초기화자(`= new ...`, `= ""`)를 유지해야 역직렬화 실패/부분 JSON 상황에서도 null 참조가 줄어든다.
- 네임스페이스를 선언하지 않는다 (프로젝트의 `HP.cs`, `GameManager.cs` 등 전역 스크립트와 동일하게 글로벌 네임스페이스).

삽입할 코드 (주석은 전부 ASCII):

```csharp

    // Phase 11 (D-03c): restore saved stats from SaveLoadManager.
    // Additive only - HP.health / HP.maxHealth are protected and PlayerStats.maxTotalHealth
    // is private with a getter-only property, so an external caller has no other way in.
    // Assignment order matters: maxHealth must be set BEFORE health, because ClampHealth()
    // clamps health into 0..maxHealth.
    public void RestoreStats(float savedHealth, float savedMaxHealth, float savedMaxTotalHealth)
    {
        maxTotalHealth = savedMaxTotalHealth;
        maxHealth = savedMaxHealth;
        health = savedHealth;
        ClampHealth();
    }
```

동작 근거:
- `health` / `maxHealth` 는 `HP` 의 `protected` 필드라 상속 클래스인 `PlayerStats` 내부에서만 대입 가능하다.
- `maxTotalHealth` 는 `PlayerStats` 자신의 `private` 필드다.
- `ClampHealth()` 는 `HP` 의 `protected` 메서드이며 `onHealthChangedCallback` 을 호출해 체력 UI 를 갱신한다 — 복원 직후 UI 가 낡은 값으로 남지 않게 하려면 반드시 호출해야 한다.
- 대입 순서(maxTotalHealth -> maxHealth -> health -> ClampHealth)를 바꾸면 저장된 health 가 낡은 maxHealth 로 잘린다.

setter 프로퍼티(`public float Health { set; }` 등)를 추가하지 않는다 — 아무나 체력을 덮어쓸 수 있게 되어 기존 데미지 파이프라인을 우회하는 구멍이 된다.
  </action>
