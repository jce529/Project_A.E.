```csharp
}

public void SpawnRandomZone()
{
    if (_zoneBounds == null) return;

    GameObject prefab = Random.value > 0.5f ? _speedUpZonePrefab : _slowDownZonePrefab;
    if (prefab == null) return;

    Bounds b = _zoneBounds.bounds;
    Vector2 pos = new Vector2(
        Random.Range(b.min.x, b.max.x),
        Random.Range(b.min.y, b.max.y));

    var zone = Instantiate(prefab, pos, Quaternion.identity);
    _activeZoneCount++;
    StartCoroutine(DestroyZoneAfter(zone, _zoneDuration));
}

private System.Collections.IEnumerator DestroyZoneAfter(GameObject zone, float duration)
{
    yield return new WaitForSeconds(duration);
    if (zone != null)
    {
        Destroy(zone);
        _activeZoneCount--;
    }
}
```

**Execute 오버라이드 추가** — 쿨다운 배율 적용 (D-03):

CombatState.Execute는 `_decisionTimer = attack.Cooldown`으로 설정한다. 광폭화 시 쿨다운을 배율로 줄이기 위해 Execute를 오버라이드하되, IBossState.Execute는 인터페이스 메서드이므로 `new` 키워드로 숨기거나 base 호출 후 배율 적용.

**방법:** CombatState.Execute 내부에서 `_decisionTimer = attack.Cooldown;` 이후에 배율을 적용할 수 없으므로 (Execute 전체를 재구현할 필요는 없음), base.Execute 호출 직후에 `_decisionTimer`에 배율을 곱하는 방식 사용.

```csharp
public new void Execute(BossController boss)
{
    base.Execute(boss);
    // 광폭화 시 쿨다운 배율 적용
    // base.Execute가 _decisionTimer = attack.Cooldown 으로 설정한 직후
    // 배율을 곱해 실질적 대기 시간 감소
    if (_isEnraged)
    {
        _decisionTimer *= _enrageCooldownMultiplier;
    }
}
```

**주의:** `CombatState`의 `Enter`, `Execute`, `Exit`는 `IBossState` 인터페이스 구현으로 virtual이 아니다. `new` 키워드로 숨김(hiding) 처리. BossController.ChangeState가 `IBossState` 인터페이스 참조로 호출하면 base 메서드가 호출되는 문제가 있을 수 있다. 하지만 WaterMonsterController.Update()에서 CombatState를 WaterMonsterCombatState로 swap하고, BossController._currentState가 실제 타입 WaterMonsterCombatState이므로, **`_currentState.Execute(this)` 호출 시 인터페이스 디스패치가 아닌 런타임 타입의 Execute가 호출된다.**

**대안 (더 안전):** `new` 대신 CombatState.Execute 끝에 virtual hook을 추가하는 방법이 있으나, 이는 CombatState.cs를 더 수정해야 한다. 실행자가 코드를 확인하고 `new` 방식이 동작하지 않으면 CombatState.Execute를 `public virtual void Execute`로 변경하는 폴백을 적용할 것.

**IBossState 인터페이스 확인 필수:** `_currentState.Execute(this)` 호출부가 인터페이스 타입(`IBossState`)으로 호출하면 `new` 메서드가 호출되지 않는다. 이 경우 **CombatState.Execute를 virtual로 변경** (CombatState.cs에서 `public void Execute` → `public virtual void Execute`)하고, WaterMonsterCombatState에서 `public override void Execute`를 사용해야 한다. 실행자는 BossController.cs의 Execute 호출부를 확인하고 적절한 방식을 선택할 것.
