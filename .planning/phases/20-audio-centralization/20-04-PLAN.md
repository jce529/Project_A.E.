```csharp
using UnityEngine;

// Phase 20 (D-13): owns the one root GameObject that survives scene loads.
//
// Why this exists: DontDestroyOnLoad only works on ROOT GameObjects. Managers that were
// scene-placed underneath a "Manager" parent (AudioManager, and one sibling manager
// affected by the same unrelated scene-transition bug, BUG-007) called DontDestroyOnLoad
// on a child, which Unity silently ignores - so they were destroyed on every scene load.
// This bootstrap creates a parentless root in code before any scene loads, so there is no
// scene file that can ever re-parent it.
//
// Extending it: add one Register<T>() line to Bootstrap(). Nothing else changes.
// Only AudioManager is registered in Phase 20; BUG-007 is deliberately NOT fixed here and
// remains a separate change. Executor note: do not name that sibling manager class in this
// file, including comments - the acceptance gate greps for its identifiers.
public static class PersistentManagers
{
    private const string RootName = "PersistentManagers";

    private static GameObject root;

    public static Transform Root => root != null ? root.transform : null;

    // BeforeSceneLoad (never the post-load variant): this must run before any scene
    // GameObject's Awake so a leftover scene-placed manager cannot win the singleton race.
    [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.BeforeSceneLoad)]
    private static void Bootstrap()
    {
        EnsureRoot();

        // --- Registered managers (Phase 20: AudioManager only) ---
        Register<AudioManager>();
    }

    private static void EnsureRoot()
    {
        if (root != null) return;

        root = new GameObject(RootName);
        // This object must stay parentless forever: a nested object cannot persist.
        Object.DontDestroyOnLoad(root);
    }

    // Idempotent. Returns the existing component if this manager is already registered,
    // so a second call (e.g. after a domain reload) never produces duplicates.
    public static T Register<T>() where T : Component
    {
        EnsureRoot();
        if (root == null) return null;

        T existing = root.GetComponent<T>();
        if (existing != null) return existing;

        return root.AddComponent<T>();
    }
}
```

Two constraints that must hold:
- There must be no `SetParent` / `transform.parent =` anywhere in this file. The whole point is that the root stays parentless.
- `Bootstrap()` must use `RuntimeInitializeLoadType.BeforeSceneLoad`, not `AfterSceneLoad`. RESEARCH.md Open Question 3 resolves this: `BeforeSceneLoad` deterministically runs before every scene `Awake()`, so even if a stale scene-placed `AudioManager` still existed somewhere, the bootstrap's instance would already hold `AudioManager.Instance` and the scene copy would hit the existing duplicate-destroy branch.

Do not add registrations for `InputHandler`, `SaveLoadManager`, or anything else. `SaveLoadManager` already has its own working bootstrap and migrating it is not requested.
