```csharp
public class BossStatsSystem : MonoBehaviour
{
    public float MaxHealth = 1000f;
    public float MaxWater = 100f;  // WARNING: 기본값 100 -> IsBarrierActive=true
    protected float _currentHealth;
    protected float _currentWater;
    public bool IsBarrierActive => _currentWater > 0;
    public event Action OnDamageTaken;
    protected void InvokeOnDamageTaken();
    void Start() { _currentHealth = MaxHealth; _currentWater = MaxWater; }
    protected virtual void TakeDamage(DamageInfo info); // override 대상
    public void TakeDamageInfo(DamageInfo info);        // 외부 진입점
    protected virtual void Die();
}
```

```csharp
public class CombatState : IBossState
{
    protected float _decisionTimer;
    public virtual void Enter(BossController boss);
    public virtual void Execute(BossController boss);  // _decisionTimer 감소, SelectAttackStrategy 호출
    protected virtual bool ShouldTransitionToGroggy(BossController boss);  // override 대상
    protected virtual IAttackStrategy SelectAttackStrategy(BossController boss, float dist);  // override 대상
    public virtual void Exit(BossController boss);
}
```
