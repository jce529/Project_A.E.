```csharp
using TMPro;
using UnityEngine;
using UnityEngine.SceneManagement;
using UnityEngine.UI;

public class MainMenuUI : MonoBehaviour
{
    [SerializeField] private Button loadGameButton;
    [SerializeField] private Color loadGameDisabledColor = new Color(0.05f, 0.05f, 0.05f, 1f);

    // Phase 14: both buttons can end up on the slot select screen, but asymmetrically -
    // "Load Game" always does (D-01), "Game Start" only when every slot is taken (D-03).
    [SerializeField] private SlotSelectPanel slotSelectPanel;
    [SerializeField] private string newGameSceneName = "Tutorial Map";

    private void Start()
    {
        if (loadGameButton != null)
        {
            // Phase 14: the guard now means "ANY slot has data", not "slot 0 has data".
            // The no-arg HasSaveFile() would only ever check CurrentSlot, which is still
            // 0 at this point - correct for the manager's internals, wrong for this button.
            SaveLoadManager mgr = SaveLoadManager.Instance;
            bool hasSave = false;
            if (mgr != null)
            {
                for (int slot = 0; slot < SaveLoadManager.SlotCount; slot++)
                {
                    if (mgr.HasSaveFile(slot))
                    {
                        hasSave = true;
                        break;
                    }
                }
            }

            loadGameButton.interactable = hasSave;

            var loadGameText = loadGameButton.GetComponentInChildren<TMP_Text>();
            if (loadGameText != null && !hasSave)
            {
                loadGameText.color = loadGameDisabledColor;
            }
        }

        if (slotSelectPanel != null)
        {
            slotSelectPanel.gameObject.SetActive(false);
        }
    }

    // "Game Start" Button OnClick.
    // D-02: first empty slot wins - start immediately, no slot screen.
    // D-03: all 3 occupied -> slot screen in new-game intent, where picking any occupied
    //       slot must clear the D-04 overwrite confirmation first.
    // This also closes a pre-existing gap: the old body called LoadScene directly and never
    // reset the save state, so starting fresh right after a load silently reused that data.
    public void OnClickStart()
    {
        SaveLoadManager mgr = SaveLoadManager.Instance;
        if (mgr == null)
        {
            Debug.LogError("[MainMenuUI] OnClickStart: SaveLoadManager.Instance is null.");
            return;
        }

        for (int slot = 0; slot < SaveLoadManager.SlotCount; slot++)
        {
            if (!mgr.HasSaveFile(slot))
            {
                mgr.NewGameInSlot(slot);
                SceneManager.LoadScene(newGameSceneName);
                return;
            }
        }

        if (slotSelectPanel == null)
        {
            Debug.LogError("[MainMenuUI] OnClickStart: every slot is occupied but slotSelectPanel is not assigned.");
            return;
        }
        slotSelectPanel.OpenForNewGame();
    }

    // "Load Game" Button OnClick.
    // D-01: always the slot screen. The player picks; nothing is ever auto-continued.
    public void OnClickLoad()
    {
        if (slotSelectPanel == null)
        {
            Debug.LogError("[MainMenuUI] OnClickLoad: slotSelectPanel is not assigned.");
            return;
        }
        slotSelectPanel.OpenForLoad();
    }
}
```
