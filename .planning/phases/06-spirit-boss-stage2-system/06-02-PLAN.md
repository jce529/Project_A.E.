```csharp
using UnityEngine;
using System.Collections;

// S2-02: 은신 + 링 범위 텔레포트 어택 전략
// D-04: 콜라이더 off → StealthDuration 대기 → 텔레포트 → 콜라이더 on
// D-05: 플레이어 중심 [MinTeleportRadius, MaxTeleportRadius] 환형 범위 내 랜덤 텔레포트
public class SpiritStealth : IAttackStrategy
{
    public float Cooldown => 2.0f;
    public string AnimationName => "";  // 로직 우선, 애니메이션은 v3.0+

    public void ExecuteAttack(BossController boss)
    {
        boss.StartCoroutine(StealthRoutine(boss));
    }

    private IEnumerator StealthRoutine(BossController boss)
    {
        if (!(boss is SpiritController spirit)) yield break;

        // 1단계: 모든 Collider2D 비활성화 (D-04a, D-04c — 피격 불가)
        Collider2D[] colliders = spirit.GetComponentsInChildren<Collider2D>(includeInactive: false);
        foreach (var c in colliders)
        {
            c.enabled = false;
        }
        Debug.Log($"[SpiritStealth] 은신 시작 — 콜라이더 {colliders.Length}개 비활성화");

        // 이동도 정지 (안전)
        spirit.StopMove();

        // 2단계: StealthDuration 동안 무적 상태로 대기
        yield return new WaitForSeconds(spirit.StealthDuration);

        // 3단계: 텔레포트 목적지 계산 (D-05: 플레이어 중심 환형 분포)
        if (spirit.Target != null)
        {
            Vector2 playerPos = spirit.Target.position;
            // 환형 분포 샘플링: 단위 원 정규화 → [Min, Max] 거리로 스케일
            Vector2 unitDir = Random.insideUnitCircle.normalized;
            if (unitDir == Vector2.zero) unitDir = Vector2.right;  // 0벡터 안전 처리
            float radius = spirit.MinTeleportRadius + Random.Range(0f, spirit.MaxTeleportRadius - spirit.MinTeleportRadius);
            Vector2 destination = playerPos + unitDir * radius;

            spirit.transform.position = destination;
            Debug.Log($"[SpiritStealth] 텔레포트 → {destination} (플레이어로부터 {radius:F2} 거리)");
        }
        else
        {
            Debug.LogWarning("[SpiritStealth] Target 이 null — 텔레포트 생략, 콜라이더만 복구");
        }

        // 4단계: 콜라이더 재활성화 (D-04a)
        foreach (var c in colliders)
        {
            if (c != null) c.enabled = true;
        }
        Debug.Log("[SpiritStealth] 은신 종료 — 콜라이더 복구");
    }
}
```

주의:
- `Random.insideUnitCircle` 은 [0,1] 범위 디스크 내부 점이므로 `.normalized` 로 단위 원으로 정규화 후 `[Min, Max]` 환형 거리를 곱해 정확한 환형 분포를 만든다 (D-05c 의 권장 방식).
- `GetComponentsInChildren<Collider2D>` 는 자식 콜라이더(HitBox 등)도 포함하므로 보스 본체뿐 아니라 자식 콜라이더까지 모두 끈다.
- `StopMove()` 호출로 텔레포트 직전 잔여 velocity 가 텔레포트 후 위치를 흐트러뜨리지 않도록 한다.
- `0벡터 안전 처리` 는 `Random.insideUnitCircle` 가 극히 드물게 (0,0) 을 반환할 경우의 NaN/0 거리 텔레포트를 방지한다.

```csharp
// D-09b/c: Stage2CombatState 가 헤비콤보 단계에서 진짜+분신 모두에게 호출.
// 진짜는 자체적으로 SpiritStealth → SpiritCharge 시퀀스를 코루틴으로 실행.
// 분신도 동일 시퀀스를 자체 실행 (분신은 데미지 0이지만 모션은 동일 — S2-04).
public void TriggerHeavyCombo()
{
    StartCoroutine(HeavyComboRoutine());
}

private System.Collections.IEnumerator HeavyComboRoutine()
{
    Debug.Log($"[SpiritController] 헤비콤보 시작 (IsDummy={IsDummy})");

    // 1단계: 은신 + 텔레포트
    var stealth = new SpiritStealth();
    stealth.ExecuteAttack(this);
    // 코루틴 완료 대기 (StealthDuration 보다 약간 긴 여유)
    yield return new UnityEngine.WaitForSeconds(StealthDuration + 0.1f);

    // 2단계: 돌진
    var charge = new SpiritCharge();
    charge.ExecuteAttack(this);
    // 돌진 완료 대기 (Windup + 최대 돌진 timeout 2.0)
    yield return new UnityEngine.WaitForSeconds(ChargeWindup + 2.0f);

    Debug.Log($"[SpiritController] 헤비콤보 종료 (IsDummy={IsDummy})");
}
```

주의:
- `StartCoroutine` 은 MonoBehaviour 메서드이므로 SpiritController(BossController:MonoBehaviour 서브클래스) 에서 직접 호출 가능하다.
- 분신도 같은 메서드를 호출받지만 IsDummy=true 이므로 SpiritCharge 의 OnTriggerEnter2D 데미지 판정에서는 어차피 PlayerStats.TakeDamage 를 호출해도 분신의 IsDummy 와 무관하게 Player 가 데미지를 받는다 (S2-04: "공격 모션을 수행하되 데미지는 0이며, 진짜 보스만 피격 데미지를 정상 적용" — 여기서 "데미지 0"은 보스가 받는 피격 데미지에 한정. 분신이 플레이어에게 주는 데미지는 별도 디자인 결정 사항이지만, 본 plan 에서는 이미 결정된 D-07 (피격 측 분기) 만 구현하고, 분신이 플레이어에게 주는 데미지는 의도적으로 동일하게 유지한다 — 분신이 플레이어를 위협하는 것이 게임 디자인의 의도).

**(B) Stage2CombatState.cs 를 본격 구현으로 교체**

Plan 01 에서 만든 스텁 파일을 다음 정확한 내용으로 완전히 교체한다:

```csharp
using UnityEngine;
using System.Collections;
using System.Collections.Generic;

// Phase 6 — Stage 2 메인 상태머신
// 사이클: 분신 생성 → 일반 패턴 N회 → 헤비콤보 (은신+돌진) → 그로기 + 분신 삭제 → (그로기 해제 시 재진입)
public class Stage2CombatState : SpiritCombatState
{
    // D-09a: 분신 SpiritController 리스트 (진짜 보스가 보유)
    private readonly List<SpiritController> _clones = new List<SpiritController>();

    // 사이클 카운터 (Discretion: N=3 회 일반 패턴 후 헤비콤보)
    private int _patternsExecuted = 0;
    private const int PatternsBeforeHeavyCombo = 3;

    // 헤비콤보 상태 추적 (Execute 가 중복 트리거하지 않도록)
    private bool _heavyComboInProgress = false;
    private float _heavyComboElapsed = 0f;
    // SpiritController.HeavyComboRoutine 의 총 소요시간 = StealthDuration + 0.1 + ChargeWindup + 2.0
    // 안전 버퍼 포함 추정 시간 (실제 종료는 _heavyComboInProgress 플래그로 추적)

    public override void Enter(BossController boss)
    {
        base.Enter(boss);
        Debug.Log("[Stage2CombatState] Stage 2 진입 — 분신 스폰 시작");

        if (!(boss is SpiritController spirit))
        {
            Debug.LogError("[Stage2CombatState] boss 가 SpiritController 가 아닙니다!");
            return;
        }

        // 분신은 Stage 2 사이클을 보유하지 않음 (분신 자신이 Stage2CombatState 진입 시도하지 않도록)
        if (spirit.IsDummy)
        {
            Debug.LogWarning("[Stage2CombatState] 분신이 Stage2CombatState 에 진입함 — SpiritCombatState 동작으로 대체");
            return;
        }

        SpawnClones(spirit);

        // 사이클 카운터 초기화
        _patternsExecuted = 0;
        _heavyComboInProgress = false;
        _heavyComboElapsed = 0f;
    }

    private void SpawnClones(SpiritController spirit)
    {
        if (spirit.DummyPrefab == null)
        {
            Debug.LogError("[Stage2CombatState] DummyPrefab 이 할당되지 않음 — 분신 스폰 실패");
            return;
        }

        // D-06: Instantiate(DummyPrefab) × 2, IsDummy=true 설정
        // 스폰 위치: 보스 좌/우 ±2 유닛 (Discretion)
        Vector3[] offsets = new Vector3[] { new Vector3(-2f, 0f, 0f), new Vector3(2f, 0f, 0f) };

        for (int i = 0; i < 2; i++)
        {
            Vector3 spawnPos = spirit.transform.position + offsets[i];
            GameObject cloneGO = Object.Instantiate(spirit.DummyPrefab, spawnPos, Quaternion.identity);

            var cloneStats = cloneGO.GetComponent<SpiritStats>();
            if (cloneStats != null)
            {
                cloneStats.IsDummy = true;  // D-06c, D-07a: 분신 표식
            }
            else
            {
                Debug.LogError($"[Stage2CombatState] 분신 #{i} 에 SpiritStats 컴포넌트 없음");
            }

            var cloneCtrl = cloneGO.GetComponent<SpiritController>();
            if (cloneCtrl != null)
            {
                _clones.Add(cloneCtrl);
                Debug.Log($"[Stage2CombatState] 분신 #{i} 스폰 완료 ({spawnPos})");
            }
            else
            {
                Debug.LogError($"[Stage2CombatState] 분신 #{i} 에 SpiritController 컴포넌트 없음");
            }
        }

        Debug.Log($"[Stage2CombatState] 분신 스폰 완료 — 총 {_clones.Count}개");
    }

    public override void Execute(BossController boss)
    {
        // 헤비콤보 진행 중이면 시간 추적만 하고 부모 Execute 는 호출 안 함
        if (_heavyComboInProgress)
        {
            _heavyComboElapsed += Time.deltaTime;

            // SpiritController.HeavyComboRoutine 총 소요시간 = StealthDuration + 0.1 + ChargeWindup + 2.0 + 여유 0.5
            if (boss is SpiritController spiritBoss)
            {
                float totalDuration = spiritBoss.StealthDuration + 0.1f + spiritBoss.ChargeWindup + 2.0f + 0.5f;
                if (_heavyComboElapsed >= totalDuration)
                {
                    OnHeavyComboFinished(boss);
                }
            }
            return;
        }

        // 일반 패턴 단계 — 부모(SpiritCombatState) 의 Execute 가 SelectAttackStrategy 호출 → 거리 기반 3종 패턴
        // 부모 Execute 가 _decisionTimer 를 관리하므로 한 사이클(공격 수행 + 쿨다운 종료) 후 다시 진입
        // 패턴 카운터는 SelectAttackStrategy 오버라이드에서 증가시킴
        base.Execute(boss);
    }

    // D-10a: SelectAttackStrategy 거리 분기는 부모(SpiritCombatState) 그대로 사용.
    //        다만 본 메서드가 호출될 때 패턴 카운터를 증가시키고, 임계 도달 시 헤비콤보로 전환.
    protected override IAttackStrategy SelectAttackStrategy(BossController boss, float dist)
    {
        // 헤비콤보 임계 도달 → 부모 호출 대신 헤비콤보 트리거
        if (_patternsExecuted >= PatternsBeforeHeavyCombo)
        {
            TriggerHeavyComboCycle(boss);
            return null;  // 부모 Execute 의 _isAttacking=true 진입을 막음 (null 반환 시 다음 프레임 재시도)
        }

        // 일반 패턴 — 부모 SpiritCombatState 의 거리 기반 분기 그대로
        var strategy = base.SelectAttackStrategy(boss, dist);
        if (strategy != null)
        {
            _patternsExecuted++;
            Debug.Log($"[Stage2CombatState] 일반 패턴 실행 #{_patternsExecuted}/{PatternsBeforeHeavyCombo} ({strategy.GetType().Name})");
        }
        return strategy;
    }

    // D-09b/c: 헤비콤보 사이클 트리거 — 진짜+분신 모두에게 TriggerHeavyCombo 호출
    private void TriggerHeavyComboCycle(BossController boss)
    {
        if (_heavyComboInProgress) return;
        _heavyComboInProgress = true;
        _heavyComboElapsed = 0f;

        Debug.Log("[Stage2CombatState] 헤비콤보 단계 진입 — 진짜+분신 동시 실행");

        if (boss is SpiritController realBoss)
        {
            realBoss.TriggerHeavyCombo();
        }

        // null 분신(이미 파괴됨)은 건너뜀
        foreach (var clone in _clones)
        {
            if (clone != null)
            {
                clone.TriggerHeavyCombo();
            }
        }
    }

    // D-08a/b: 헤비콤보 종료 → 분신 전체 Destroy → GroggyState 전환
    private void OnHeavyComboFinished(BossController boss)
    {
        Debug.Log("[Stage2CombatState] 헤비콤보 종료 — 분신 삭제 + 그로기 전환");

        // 분신 전체 Destroy
        foreach (var clone in _clones)
        {
            if (clone != null && clone.gameObject != null)
            {
                Object.Destroy(clone.gameObject);
            }
        }
        _clones.Clear();

        _heavyComboInProgress = false;

        // 그로기 전환 — 5초 후 GroggyState 가 plain CombatState 로 전환 → SpiritController.Update 인터셉트가
        // IsStage2=true 이므로 Stage2CombatState 새 인스턴스로 교체 → Enter() 에서 분신 재스폰 (사이클 반복)
        boss.ChangeState(new GroggyState());
    }

    public override void Exit(BossController boss)
    {
        base.Exit(boss);
        // Exit 에서 분신을 직접 정리하지 않음 (정상 종료는 OnHeavyComboFinished 에서 처리됨)
        // 비정상 Exit (예: Die) 시 분신 잔존 가능 — 보스 사망 시점에 SpiritStats.Die 에서 처리하는 것이 적절하지만
        // 본 plan 범위에서는 GroggyState 전환만 정상 흐름으로 가정.
    }

    // ShouldTransitionToGroggy 는 부모(SpiritCombatState) 가 false 반환 — Stage2 도 그로기 자동 전환 없이
    // OnHeavyComboFinished 의 명시적 ChangeState(new GroggyState()) 만 사용
}
```
