신규 파일 `Assets/Enemy/WaterMonster/Script/Phase3/WaterTeleportState.cs` 생성 (per D-07, D-08, D-09, D-10, D-11, D-15):

```csharp
using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using WaterMonster.Phase2;

/// <summary>
/// WaterTeleportState — IBossState 구현.
/// 보스가 파괴 불가 웅덩이로 순간이동하는 농락형 포지셔닝 패턴.
/// D-07: 플레이어와의 거리에 반비례하여 타겟 웅덩이 선택.
/// D-11: REQ-WM-02 HP 코스트 적용.
/// D-15: Enter에서 타겟 선택 + HP 코스트 + 연출 코루틴 시작, 완료 후 CombatState 복귀.
/// </summary>
public class WaterTeleportState : IBossState
{
    private const float HpCostPercent = 0.05f;       // 텔레포트 HP 코스트 (MaxHealth의 5%)
    private const float CloseRangeThreshold = 3.0f;  // D-08: CombatState의 dist <= 3.0f 기준 재사용
    private const float TeleportDelay = 0.25f;       // D-09: 사라짐/나타남 사이 딜레이 (0.2~0.3초 범위)

    private Coroutine _routine;

    public void Enter(BossController boss)
    {
        // D-11, REQ-WM-02: HP 코스트 차감
        if (boss.Stats is WaterMonsterStats wms)
        {
            wms.SpendHpCost(wms.MaxHealth * HpCostPercent);
        }

        // D-07: 타겟 웅덩이 선택
        WaterPuddle target = SelectTeleportTarget(boss);
        if (target == null)
        {
            // 웅덩이 없으면 즉시 CombatState 복귀
            boss.ChangeState(new WaterMonsterCombatState());
            return;
        }

        // 텔레포트 쿨다운 기록 (WaterMonsterController에 저장)
        if (boss is WaterMonsterController wmc)
        {
            wmc.RecordTeleportTime();
        }

        // 연출 코루틴 시작 — IBossState는 MonoBehaviour가 아니므로 boss 경유
        _routine = boss.StartCoroutine(TeleportSequence(boss, target));
    }

    public void Execute(BossController boss)
    {
        // 코루틴이 모든 처리를 수행하므로 빈 구현
    }

    public void Exit(BossController boss)
    {
        // Pitfall 2 방지: 외부 State 강제 교체 시 코루틴 정리
        if (_routine != null)
        {
            boss.StopCoroutine(_routine);
            _routine = null;
        }
        // 사라짐 VFX로 비활성된 SpriteRenderer 복원
        var sr = boss.GetComponent<SpriteRenderer>();
        if (sr != null) sr.enabled = true;
    }

    private IEnumerator TeleportSequence(BossController boss, WaterPuddle target)
    {
        var sr = boss.GetComponent<SpriteRenderer>();

        // D-09: 출발지 사라짐 VFX — SpriteRenderer.enabled = false (에셋 없이 구현)
        if (sr != null) sr.enabled = false;

        yield return new WaitForSeconds(TeleportDelay);

        // position 이동
        boss.transform.position = target.transform.position;

        // D-09: 목적지 나타남 VFX — SpriteRenderer.enabled = true
        if (sr != null) sr.enabled = true;

        yield return new WaitForSeconds(TeleportDelay);

        _routine = null;

        // D-10: 도착 직후 즉시 WaterMonsterCombatState 복귀
        // 중요: new CombatState()가 아닌 new WaterMonsterCombatState() (스왑 루프 방지, RESEARCH Anti-Pattern 참조)
        boss.ChangeState(new WaterMonsterCombatState());
    }

    /// <summary>
    /// D-07: 플레이어와의 거리에 반비례하여 타겟 웅덩이 선택.
    /// 가까이 (dist <= 3.0f) → 플레이어에서 가장 먼 웅덩이
    /// 멀리 (dist > 3.0f) → 플레이어에서 가장 가까운 웅덩이
    /// </summary>
    private WaterPuddle SelectTeleportTarget(BossController boss)
    {
        if (PuddleStackManager.Instance == null) return null;

        var puddles = PuddleStackManager.Instance.IndestructiblePuddles;
        if (puddles == null || puddles.Count == 0) return null;

        float distToPlayer = Vector2.Distance(boss.transform.position, boss.Target.position);
        bool isClose = distToPlayer <= CloseRangeThreshold; // D-08

        WaterPuddle selected = null;
        float bestDist = isClose ? float.MinValue : float.MaxValue;

        foreach (var puddle in puddles)
        {
            if (puddle == null || !puddle.gameObject.activeSelf) continue;

            float d = Vector2.Distance(boss.Target.position, puddle.transform.position);

            if (isClose)
            {
                // 가까이 → 플레이어에서 가장 먼 웅덩이
                if (d > bestDist) { bestDist = d; selected = puddle; }
            }
            else
            {
                // 멀리 → 플레이어에서 가장 가까운 웅덩이
                if (d < bestDist) { bestDist = d; selected = puddle; }
            }
        }

        return selected;
    }
}
```

주요 설계 포인트:
- `HpCostPercent = 0.05f` — 텔레포트는 근접 공격(0.03f)보다 약간 높은 코스트. 인스펙터 튜닝 가능하게 하려면 const 대신 별도 필드로 변환 가능하나, IBossState는 plain class이므로 const로 유지.
- `CloseRangeThreshold = 3.0f` — D-08, CombatState.SelectAttackStrategy의 `dist <= 3.0f` 기준 재사용.
- `TeleportDelay = 0.25f` — D-09 범위(0.2~0.3초) 내 중간값.
- `Exit()`에서 `StopCoroutine` + SpriteRenderer 복원 — Pitfall 2 방지.
- 텔레포트 완료 후 `new WaterMonsterCombatState()` 사용 (CombatState 아님) — RESEARCH Anti-Pattern 방지.
- `RecordTeleportTime()` 호출로 쿨다운을 WaterMonsterController에 기록 (Task 2에서 추가).

**파일 1: WaterMonsterCombatState.cs 수정** (per D-12, D-13, D-14):

기존 파일 전체를 다음으로 교체한다:

```csharp
// Phase 1+3 — WaterMonsterCombatState
// Phase 1: ShouldTransitionToGroggy override (MaxWater=0 대응)
// Phase 3: SelectAttackStrategy에 텔레포트 조건 분기 추가 (D-12, D-13)

using WaterMonster.Phase2;

public class WaterMonsterCombatState : CombatState
{
    protected override bool ShouldTransitionToGroggy(BossController boss)
    {
        return false;
    }

    protected override IAttackStrategy SelectAttackStrategy(BossController boss, float dist)
    {
        // D-13: 텔레포트 선택 조건 — Indestructible >= 2 이고 쿨다운 경과
        bool canTeleport = PuddleStackManager.Instance != null
            && PuddleStackManager.Instance.IndestructibleCount >= 2
            && boss is WaterMonsterController wmc
            && wmc.CanTeleport();

        if (canTeleport)
        {
            // D-12: SelectAttackStrategy 내에서 텔레포트를 패턴 후보로 통합
            // ChangeState 후 null 반환 — CombatState.Execute()의 기존 if(attack != null) 가드가
            // 이후 공격 실행을 막아줌 (RESEARCH Pitfall 4 참조)
            boss.ChangeState(new WaterTeleportState());
            return null;
        }

        // D-08: 기존 패턴 (dist <= 3.0f 근접, 그 외 원거리)
        if (dist <= 3.0f)
        {
            return new WaterMeleeSwipe();
        }
        return new WaterRangedSpit();
    }
}
```
