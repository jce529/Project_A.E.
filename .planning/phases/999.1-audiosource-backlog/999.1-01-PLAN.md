    // D-03: the only SFX entry points. One-shot only (no SFX loops this phase).
    public bool Play(AudioCue cue)
    {
        return PlayInternal(cue, false, Vector3.zero, null);
    }

    public bool PlayAt(AudioCue cue, Vector3 worldPosition)
    {
        return PlayInternal(cue, true, worldPosition, null);
    }

    public bool PlayFollow(AudioCue cue, Transform target)
    {
        if (target == null) return false;
        return PlayInternal(cue, true, target.position, target);
    }

    private bool PlayInternal(AudioCue cue, bool positional, Vector3 position, Transform follow)
    {
        if (cue == null || cue.clips == null || cue.clips.Length == 0) return false;
        AudioClip clip = cue.clips[Random.Range(0, cue.clips.Length)];
        if (clip == null) return false;

        // D-05: per-cue retrigger cooldown, unscaled so the pause menu (timeScale 0) does not freeze it.
        float now = Time.unscaledTime;
        float last;
        if (cue.retriggerCooldown > 0f && lastPlayTime.TryGetValue(cue, out last) && now - last < cue.retriggerCooldown)
            return false;

        // D-05: per-cue concurrency limit (0 = unlimited). Over the limit the new request is rejected.
        if (cue.maxConcurrent > 0 && CountActive(cue) >= cue.maxConcurrent) return false;

        int index = AcquireVoice(cue.priority);
        if (index < 0) return false;

        // Pooled voices are reused: set every property on every play (no stale pitch/spatial state).
        Voice v = voices[index];
        AudioSource src = v.src;
        src.clip = clip;
        src.loop = false;
        src.pitch = Random.Range(cue.pitchMin, cue.pitchMax);
        v.baseVolume = cue.volume;
        src.volume = cue.volume * SfxVolume;
        if (positional)
        {
            src.spatialBlend = 1f;
            src.rolloffMode = AudioRolloffMode.Linear;
            src.minDistance = cue.minDistance;
            src.maxDistance = cue.maxDistance;
            src.dopplerLevel = 0f;
            src.spread = 0f;
            src.transform.position = ToListenerPlane(position);
        }
        else
        {
            src.spatialBlend = 0f;
            src.transform.localPosition = Vector3.zero;
        }

        v.cue = cue;
        v.priority = cue.priority;
        v.startTime = now;
        v.follow = follow;
        v.active = true;
        lastPlayTime[cue] = now;
        src.Play();
        return true;
    }

    private int CountActive(AudioCue cue)
    {
        int count = 0;
        for (int i = 0; i < voices.Length; i++)
        {
            if (voices[i].active && voices[i].cue == cue) count++;
        }
        return count;
    }

    // D-04/D-06: free voice first; otherwise steal the lowest priority, then the oldest.
    // A voice with higher priority than the incoming request is never stolen.
    private int AcquireVoice(AudioPriority incoming)
    {
        int lowest = -1;
        for (int i = 0; i < voices.Length; i++)
        {
            if (!voices[i].active) return i;
            if (lowest < 0 || voices[i].priority < voices[lowest].priority ||
                (voices[i].priority == voices[lowest].priority && voices[i].startTime < voices[lowest].startTime))
                lowest = i;
        }
        if (voices[lowest].priority > incoming) return -1;
        voices[lowest].src.Stop();
        voices[lowest].active = false;
        return lowest;
    }

    // 2D: the AudioListener is on Main Camera (z = -10 in gameplay scenes) while gameplay is at z = 0.
    // Placing sources at the listener's z makes only XY distance attenuate.
    private static Vector3 ToListenerPlane(Vector3 p)
    {
        Camera cam = Camera.main;
        float z = cam != null ? cam.transform.position.z : 0f;
        return new Vector3(p.x, p.y, z);
    }
